\begin{card}[equal height group=evid]{Evidência 3 \textbullet\ Melhorias nos Agentes de IA}
{\footnotesize
\textbf{EVOLUÍDO} \ Refinamento do Assistente ReAct (LangGraph):\vspace{2pt}
\begin{itemize}
  \item \textbf{Edição inline:} ao clicar em \textit{Editar}, o analista ajusta o texto exato que segue para a análise final com opção de \textit{Desfazer}.
  \item \textbf{Planos em 3 níveis:} tarefas classificadas em \textit{Contenção, Corretiva e Preventiva} com decisão individual (\textit{Aceitar/Editar/Rejeitar}).
  \item \textbf{Casos parecidos:} busca semântica explicando o motivo da similaridade (\textit{mesmo tipo, mesma linha, texto parecido}).
\end{itemize}}
